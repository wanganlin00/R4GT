brackets <- data.frame(
  x    = c(0.75, 0.75, 1.25, 1.75, 1.75, 2.75),   # 每段起点 x
  xend = c(1.25, 0.75, 1.25, 2.75, 1.75, 2.75),   # 每段终点 x
  y    = c(20,   20,   20,   35,   35,   35),      # 每段起点 y
  yend = c(20,   19.2, 19.2, 35,   34.2, 34.2)     # 每段终点 y（竖线向下）
)

p + geom_segment(data = brackets,

library(rstatix); library(ggpubr)

stat.test <- df %>%
  group_by(dose) %>%
  t_test(len ~ supp) %>%        # 每组内的 OJ vs VC
  add_significance()

stat.test <- add_xy_position(stat.test, x = "dose", dodge = 1)
# 自动得到（与手动 layer_data 结果完全一致）：
#   dose  xmin  xmax  y.position  p.signif
#   0.5   0.75  1.25  24.2        **
#   1     1.75  2.25  30.0        **
#   2     2.75  3.25  36.6        ns

p + stat_pvalue_manual(stat.test, label = "p", tip.length = 0.015)
